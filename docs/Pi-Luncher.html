Pi Agent 批量自动化启动工具

一个用 Rust 编写的启动器工具，一条命令完成 环境预检 → pi update --all → 检查并升级 pi-web → 启动 pi-web → 启动 Pi CLI 的完整流程。每个环节都有状态显示与错误检查；支持安静模式（自动化 / 计划任务）与全程文字日志。

v1.0.0 Rust 2021 indicatif 0.17 脱离式后台 --quiet 安静模式 全程文字日志

use std::io::{self, Read, Write};
use std::net::{TcpStream, ToSocketAddrs};
use std::process::{Child, Command, Stdio};
use std::thread;
use std::time::{Duration, Instant};

/// 命令执行结果
#[derive(Debug)]
pub enum RunResult {
    /// 命令成功完成
    Success,
    /// 命令失败，包含退出码和 stderr 摘要
    Failure { code: i32, stderr: String },
    /// 命令未找到（可执行文件不存在）
    NotFound(String),
    /// 启动或等待过程中发生 IO 错误
    IoError(String),
}

/// 服务起動探测结果
#[derive(Debug)]
pub enum StartupOutcome {
    /// 进程存活且已能回应 HTTP
    Up,
    /// 进程在就绪前就退出了
    Exited,
    /// 超时仍未就绪
    Timeout,
}

/// 检查命令是否存在于 PATH 中
pub fn command_exists(name: &str) -> bool {
    let check_cmd = if cfg!(target_os = "windows") {
        "where"
    } else {
        "which"
    };

    Command::new(check_cmd)
        .arg(name)
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status()
        .map(|s| s.success())
        .unwrap_or(false)
}

/// 解析要启动的程序。
///
/// Windows 上 npm 全局命令（如 `pi`、`pi-web`）实际是 `.cmd` / `.bat` shim，
/// `CreateProcess` 无法直接执行它们（会得到 NotFound），必须经 `cmd.exe` 包装。
/// 返回 `(真实路径, 是否需要经 cmd 包装)`：
/// - 找到 `.exe` → 直接启动
/// - 找到 `.cmd` / `.bat` → 用 `cmd /C` 启动
/// - 找不到 → 原样交给 `cmd /C` 解析
#[cfg(windows)]
fn resolve_launch(program: &str) -> (String, bool) {
    if let Ok(output) = Command::new("where").arg(program).output() {
        if output.status.success() {
            let text = String::from_utf8_lossy(&output.stdout);
            let mut exe = None;
            let mut script = None;
            for line in text.lines() {
                let p = line.trim();
                if p.is_empty() {
                    continue;
                }
                let lower = p.to_ascii_lowercase();
                if lower.ends_with(".exe") && exe.is_none() {
                    exe = Some(p.to_string());
                } else if (lower.ends_with(".cmd") || lower.ends_with(".bat")) && script.is_none() {
                    script = Some(p.to_string());
                }
            }
            if let Some(e) = exe {
                return (e, false);
            }
            if let Some(s) = script {
                return (s, true);
            }
        }
    }
    (program.to_string(), true)
}

#[cfg(not(windows))]
fn resolve_launch(program: &str) -> (String, bool) {
    (program.to_string(), false)
}

/// 构造一个 `Command`，统一处理跨平台（尤其是 Windows `.cmd` shim）的启动差异。
pub fn build_command(program: &str) -> Command {
    let (real, via_cmd) = resolve_launch(program);
    if via_cmd {
        let mut cmd = Command::new("cmd.exe");
        cmd.arg("/C").arg(real);
        cmd
    } else {
        Command::new(real)
    }
}

/// 命令的原始输出（stdout / stderr 与退出码），供需要读取输出内容的场景使用。
#[derive(Debug)]
pub struct RawOutput {
    pub success: bool,
    pub code: i32,
    pub stdout: String,
    pub stderr: String,
}

/// 运行一个前台命令并捕获其输出，本身不判定成败；
/// 仅当命令无法启动（未找到 / IO 错误）时返回 Err。
pub fn run_capture(program: &str, args: &[&str]) -> Result<RawOutput, RunResult> {
    let output = match build_command(program)
        .args(args)
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .output()
    {
        Ok(o) => o,
        Err(e) if e.kind() == io::ErrorKind::NotFound => {
            return Err(RunResult::NotFound(program.to_string()));
        }
        Err(e) => return Err(RunResult::IoError(e.to_string())),
    };

    Ok(RawOutput {
        success: output.status.success(),
        code: output.status.code().unwrap_or(-1),
        stdout: String::from_utf8_lossy(&output.stdout)
            .trim_end()
            .to_string(),
        stderr: String::from_utf8_lossy(&output.stderr)
            .lines()
            .rev()
            .take(5)
            .collect::<Vec<_>>()
            .join("\n"),
    })
}

/// 运行一个前台命令，等待其完成，并检查退出状态。
/// stdout/stderr 会被捕获，失败时提取摘要。
pub fn run_checked(program: &str, args: &[&str]) -> RunResult {
    match run_capture(program, args) {
        Ok(o) if o.success => RunResult::Success,
        Ok(o) => RunResult::Failure {
            code: o.code,
            stderr: o.stderr,
        },
        Err(e) => e,
    }
}

/// 以后台方式启动一个常驻服务，并使其脱离本进程 / 控制台。
///
/// 关键点：父进程（launcher）退出、或终端视窗被关闭时，子进程不会被连带终止。
/// - Windows：`DETACHED_PROCESS | CREATE_NEW_PROCESS_GROUP`，不附着任何 console，
///   因此 Ctrl+C / 关窗都不会波及它。
/// - Unix：`process_group(0)` 另开进程组。
pub fn spawn_detached(program: &str, args: &[&str]) -> Result<Child, String> {
    let mut cmd = build_command(program);
    cmd.args(args)
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::null());

    #[cfg(windows)]
    {
        use std::os::windows::process::CommandExt;
        const CREATE_NEW_PROCESS_GROUP: u32 = 0x0000_0200;
        const DETACHED_PROCESS: u32 = 0x0000_0008;
        cmd.creation_flags(CREATE_NEW_PROCESS_GROUP | DETACHED_PROCESS);
    }
    #[cfg(unix)]
    {
        use std::os::unix::process::CommandExt;
        cmd.process_group(0);
    }

    cmd.spawn()
        .map_err(|e| format!("无法启动 {}: {}", program, e))
}

/// 对 `host:port` 做一次最小 HTTP 探测，确认服务真的在回应（而不只是端口开着）。
pub fn probe_http(host: &str, port: u16) -> bool {
    let addr = match (host, port).to_socket_addrs() {
        Ok(mut it) => match it.next() {
            Some(a) => a,
            None => return false,
        },
        Err(_) => return false,
    };

    let mut stream = match TcpStream::connect_timeout(&addr, Duration::from_millis(500)) {
        Ok(s) => s,
        Err(_) => return false,
    };
    let _ = stream.set_read_timeout(Some(Duration::from_millis(500)));
    let _ = stream.set_write_timeout(Some(Duration::from_millis(500)));

    let request = format!(
        "GET / HTTP/1.0\r\nHost: {}\r\nConnection: close\r\n\r\n",
        host
    );
    if stream.write_all(request.as_bytes()).is_err() {
        return false;
    }

    let mut buf = [0u8; 16];
    match stream.read(&mut buf) {
        Ok(n) if n >= 5 => buf.starts_with(b"HTTP/"),
        _ => false,
    }
}

/// 等待服务真正就绪：既确认进程没有退出，也确认 HTTP 端口能回应。
/// progress：可选回调，按等待进度触发（0..1），供确定进度条爬升使用。
pub fn wait_ready<P: FnMut(f64)>(
    child: &mut Child,
    host: &str,
    port: u16,
    timeout_secs: u64,
    mut progress: Option<P>,
) -> StartupOutcome {
    let deadline = Instant::now() + Duration::from_secs(timeout_secs);
    let started = Instant::now();
    loop {
        if let Ok(Some(_status)) = child.try_wait() {
            return StartupOutcome::Exited;
        }
        if probe_http(host, port) {
            if let Some(cb) = progress.as_mut() {
                cb(1.0);
            }
            return StartupOutcome::Up;
        }
        if let Some(cb) = progress.as_mut() {
            let frac = (started.elapsed().as_secs_f64() / timeout_secs as f64).min(1.0);
            cb(frac);
        }
        if Instant::now() >= deadline {
            return StartupOutcome::Timeout;
        }
        thread::sleep(Duration::from_millis(250));
    }
}

/// 按 PID 终止进程及其整棵进程树。
///
/// 后台 pi-web 经 `cmd /C` 启动，真正的服务是其子进程（node），
/// 只 kill 父进程会留下孤儿进程占用端口，因此需要连同子进程一起终止。
pub fn terminate_pid(pid: u32) {
    #[cfg(windows)]
    {
        let _ = Command::new("taskkill")
            .args(["/PID", &pid.to_string(), "/T", "/F"])
            .stdout(Stdio::null())
            .stderr(Stdio::null())
            .status();
    }
    #[cfg(not(windows))]
    {
        let _ = Command::new("kill")
            .arg("-TERM")
            .arg(pid.to_string())
            .stdout(Stdio::null())
            .stderr(Stdio::null())
            .status();
    }
}

/// 终止进程树并回收句柄（用于启动失败 / 超时的半途进程）。
pub fn terminate_background(child: &mut Child) {
    terminate_pid(child.id());
    let _ = child.wait();
}

/// 步骤 1：环境预检（pi / pi-web / npm）
pub fn step_preflight(ui: &mut Ui) -> Result<(), String> {
    ui.step_start(1, "环境预检（pi / pi-web / npm）", false);

    let required = ["pi", "pi-web", "npm"];
    for name in &required {
        ui.step_message(&format!("正在检查命令 `{}` …", name));
        if !command_exists(name) {
            let msg = format!(
                "未找到命令 `{}`。请先安装 Pi Coding Agent：\n   npm install -g --ignore-scripts @earendil-works/pi-coding-agent",
                name
            );
            ui.step_fail(&format!("未找到命令 `{}`", name));
            ui.message(
                "    需先执行：npm install -g --ignore-scripts @earendil-works/pi-coding-agent",
            );
            return Err(msg);
        }
        ui.message(&format!("  ✓ 命令 `{}` 已就绪", name));
    }

    ui.step_ok("pi / pi-web / npm 全部就绪");
    Ok(())
}

/// 步骤 2：执行 pi update --all
pub fn step_pi_update(ui: &mut Ui) -> Result<(), String> {
    ui.step_start(2, "执行 pi update --all", false);
    ui.step_message("正在更新 Pi 及扩展包…");

    let result = run_checked("pi", &["update", "--all"]);

    match result {
        RunResult::Success => {
            ui.step_ok("Pi 及所有扩展包更新完成");
            Ok(())
        }
        RunResult::NotFound(prog) => {
            let msg = format!("{} 命令未找到，请检查安装", prog);
            ui.step_fail(&msg);
            Err(msg)
        }
        RunResult::Failure { code, stderr } => {
            ui.step_fail(&format!("pi update 失败（退出码 {}）", code));
            ui.print_stderr_summary(&stderr);
            Err("pi update 执行失败，启动中止".to_string())
        }
        RunResult::IoError(e) => {
            let msg = format!("执行 pi update 时发生 IO 错误: {}", e);
            ui.step_fail(&msg);
            Err(msg)
        }
    }
}

/// 步骤 3：检查 pi-web 是否有新版本，有则升级（npm install -g @agegr/pi-web@latest）。
///
/// 流程：
/// 0. pi-web 正在运行 → 跳过升级（进程锁定文件，npm -g 替换会 EBUSY 失败）
/// 1. `npm view @agegr/pi-web version` 查询 registry 最新版
/// 2. `npm ls -g @agegr/pi-web` 查询本地已装版本
/// 3. 本地较旧（或本地未装 / 本地查询失败）→ 执行升级安装命令
///
/// 任何查询失败都视为“无法确认”，走安全路径（直接执行安装命令由 npm 自行判断）。
pub fn step_pi_web_upgrade(ui: &mut Ui, host: &str, port: u16) -> Result<(), String> {
    ui.step_start(3, "检查并升级 pi-web", false);

    // 运行中的实例会锁定包目录，npm -g 升级必然失败（EBUSY），
    // 因此先探测：在运行就跳过升级，保留现有版本。
    if probe_http(host, port) {
        ui.step_ok(&format!(
            "pi-web 正在运行（http://{}:{}），跳过升级",
            host, port
        ));
        ui.message(
            "  ⚠ 运行中的 pi-web 会锁定文件，npm -g 升级无法替换；如需升级：先执行 `pi-web stop`，再运行 launcher。",
        );
        return Ok(());
    }

    ui.step_message("查询 registry 最新版本…");
    let remote = run_capture("npm", &["view", PI_WEB_PKG, "version"]);

    if let Err(RunResult::NotFound(prog)) = &remote {
        let msg = format!("{} 命令未找到，请检查安装", prog);
        ui.step_fail(&msg);
        return Err(msg);
    }

    let remote_ver = remote
        .ok()
        .map(|o| o.stdout)
        .as_deref()
        .and_then(extract_version);
    ui.step_message("查询本地已安装版本…");
    let local = run_capture("npm", &["ls", "-g", PI_WEB_PKG]);
    let local_ok = local.is_ok();
    let local_ver = local.ok().and_then(|o| {
        if o.success {
            extract_version(&o.stdout)
        } else {
            // npm ls 反查不存在的包时非零退出，视为「本地未装由 npm 处理」
            None
        }
    });

    match (remote_ver.as_deref(), local_ver.as_deref()) {
        (Some(r), Some(l)) if compare_versions(l, r) != std::cmp::Ordering::Less => {
            ui.step_ok(&format!("pi-web 已是最新版本（v{}）", l));
            Ok(())
        }
        (Some(r), Some(l)) => {
            ui.message(&format!("  ⚠ 发现新版本：v{} → v{}，正在升级…", l, r));
            run_upgrade_install(ui)
        }
        // 本地版本查不到 / 未安装：无法确认，直接跑安装命令让 npm 判断
        (Some(_), None) => {
            if local_ok {
                ui.message("  ⚠ 本地已安装但无法解析版本号，直接执行升级命令…");
            } else {
                ui.message("  ⚠ pi-web 未以全局包形式安装，直接执行安装…");
            }
            run_upgrade_install(ui)
        }
        // registry 查询失败：网络问题或包不存在，同样兜底直接安装
        (None, _) => {
            ui.message("  ⚠ 无法查询 registry 上的最新版本，回退为直接执行升级命令…");
            run_upgrade_install(ui)
        }
    }
}

/// 执行 `npm install -g @agegr/pi-web@latest` 并处理结果。
fn run_upgrade_install(ui: &mut Ui) -> Result<(), String> {
    ui.step_message(&format!("npm install -g {} …", PI_WEB_PKG_LATEST));

    let result = run_checked("npm", &["install", "-g", PI_WEB_PKG_LATEST]);

    match result {
        RunResult::Success => {
            // 复查一次实际安装到的版本，确认结果
            let now = run_capture("npm", &["ls", "-g", PI_WEB_PKG])
                .ok()
                .filter(|o| o.success)
                .and_then(|o| extract_version(&o.stdout))
                .unwrap_or_default();
            if now.is_empty() {
                ui.step_ok("pi-web 升级完成");
            } else {
                ui.step_ok(&format!("pi-web 升级完成，当前版本 v{}", now));
            }
            Ok(())
        }
        RunResult::NotFound(prog) => {
            let msg = format!("{} 命令未找到，请检查安装", prog);
            ui.step_fail(&msg);
            Err(msg)
        }
        RunResult::Failure { code, stderr } => {
            // pi-web 常驻实例会锁定包目录，npm -g 无法替换（EBUSY / busy or locked）。
            // 这类失败不中止启动：按现有版本继续，提示先 stop 后再跑一次 launcher 完成升级。
            let locked =
                code == -4082 || stderr.contains("EBUSY") || stderr.contains("busy or locked");
            if locked {
                ui.step_ok("pi-web 升级被运行中的实例文件锁挡下，按现有版本继续启动");
                ui.message("  ⚠ 如需完成升级：先执行 `pi-web stop`，再运行一次 launcher。");
                ui.message("    （npm 错误详情已记录到日志）");
                return Ok(());
            }
            ui.step_fail(&format!("pi-web 升级失败（退出码 {}）", code));
            ui.print_stderr_summary(&stderr);
            Err("pi-web 升级失败，启动中止".to_string())
        }
        RunResult::IoError(e) => {
            let msg = format!("执行 pi-web 升级时发生 IO 错误: {}", e);
            ui.step_fail(&msg);
            Err(msg)
        }
    }
}

/// 步骤 4：启动 pi-web（常驻后台服务，脱离本进程）。
///
/// - 启动前先 HTTP 探测：服务已在运行则跳过（保留原进程，本进程无从监控）
/// - 否则 detached 启动，并以确定进度条展示「等待 HTTP 就绪」的时间轴（60s 上限）
pub fn step_pi_web(ui: &mut Ui, host: &str, port: u16) -> Result<WebStart, String> {
    ui.step_start(4, "启动 pi-web（常驻后台）", true);

    // 已在运行？直接跳过，避免端口冲突
    if probe_http(host, port) {
        let msg = format!(
            "pi-web 已在运行：http://{}:{}，跳过启动（保留原进程，本次不做崩溃监控）",
            host, port
        );
        ui.step_ok(&msg);
        return Ok(WebStart::AlreadyRunning);
    }

    ui.step_message("正在启动 pi-web 服务…");

    let mut child = match spawn_detached("pi-web", &[]) {
        Ok(c) => c,
        Err(e) => {
            ui.step_fail(&format!("pi-web 启动失败：{}", e));
            return Err(e);
        }
    };

    ui.step_message(&format!(
        "等待 HTTP 就绪 http://{}:{}（上限 {}s）…",
        host, port, READY_TIMEOUT_SECS
    ));

    let progress = |frac: f64| {
        ui.step_progress(frac);
    };

    match wait_ready(&mut child, host, port, READY_TIMEOUT_SECS, Some(progress)) {
        StartupOutcome::Up => {
            ui.step_ok(&format!("pi-web 已启动并就绪：http://{}:{}", host, port));
            ui.message("  ⚠ pi-web 已脱离本进程，launcher / CLI 退出后仍会继续运行");
            ui.message("  ⚠ 如需停止，请执行：pi-web stop");
            Ok(WebStart::Spawned(child))
        }
        StartupOutcome::Exited => {
            ui.step_fail("pi-web 进程在就绪前就退出了，可能端口被占用或配置有误");
            Err("pi-web 启动失败".to_string())
        }
        StartupOutcome::Timeout => {
            terminate_background(&mut child);
            ui.step_fail(&format!("等待 pi-web 就绪超时（{}s）", READY_TIMEOUT_SECS));
            Err("pi-web 启动超时".to_string())
        }
    }
}

/// 交互模式是在「stdin/stdout 都是 TTY 且非 --print」时的预设行为，所以直接跑 `pi`。

/// 比较两个 semver 版本号（如 "1.2.3"），相等返回 Ordering::Equal，
/// 逐段按数值比较；无法解析的部分按数值 0 处理。
fn compare_versions(a: &str, b: &str) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    let parse_num = |part: &str| -> u64 { part.parse().unwrap_or(0) };
    let mut pa = a.split('.');
    let mut pb = b.split('.');

    loop {
        match (pa.next(), pb.next()) {
            (None, None) => return Ordering::Equal,
            (x, y) => {
                let nx = x.map(parse_num).unwrap_or(0);
                let ny = y.map(parse_num).unwrap_or(0);
                match nx.cmp(&ny) {
                    Ordering::Equal => continue,
                    other => return other,
                }
            }
        }
    }
}

/// 从字符串中提取形如 x.y.z 的版本号。
/// 兼容 `npm view` 的纯版本输出与 `npm ls -g` 的 `├── pkg@1.2.3` 树状输出。
fn extract_version(text: &str) -> Option<String> {
    for raw_line in text.lines() {
        let line = raw_line.trim();

        // 1) `pkg@x.y.z` 树状输出：取包名后最后一个 '@' 之后的内容
        if let Some(i) = line.find(PI_WEB_PKG) {
            if let Some(off) = line[i..].rfind('@') {
                let candidate = line[i + off + 1..].trim();
                if is_version(candidate) {
                    return Some(candidate.to_string());
                }
            }
            continue;
        }

        // 2) `npm view ... version` 的裸版本号输出（如 `0.4.2`）
        if is_version(line) {
            return Some(line.to_string());
        }
    }
    None
}

/// 判断是否为合法版本号片段（如 `0.4.2`、`1.0.0-rc.1`、`2.0.0+build`）。
fn is_version(s: &str) -> bool {
    !s.is_empty()
        && s.chars()
            .all(|c| c.is_ascii_alphanumeric() || c == '.' || c == '-' || c == '+')
}

//! 统一输出层：控制台渲染 + 全程文字日志。
//!
//! 所有步骤的打印与日志必须经由本模块，保证：
//! - 交互模式（TTY 进度动画）：步骤行「左消息 + 右侧固定位置动画/进度条」，
//!   底部常驻整体进度；普通行输出用 MultiProgress::println，避免与重绘打架；
//! - 非 TTY（管道/CI）：纯文字退化（println），无渲染开销；
//! - 安静模式（--quiet）：控制台零输出，但日志照常记录。
//!

static LOGGER: OnceLock<Mutex<Logger>> = OnceLock::new();

struct Logger {
    path: PathBuf,
    file: BufWriter<File>,
}

impl Logger {
    fn new() -> io::Result<Self> {
        let dir = log_dir();
        fs::create_dir_all(&dir)?;
        cleanup_logs(&dir, LOG_KEEP)?;

        let path = dir.join(format!("{}{}.log", LOG_PREFIX, stamp_hex()));
        let file = File::create(&path)?;
        let mut writer = BufWriter::new(file);
        // UTF-8 BOM：让 PowerShell / 旧版记事本正确识别编码
        let _ = writer.write_all(&[0xEF, 0xBB, 0xBF]);
        Ok(Self { path, file: writer })
    }

    fn write(&mut self, msg: &str) {
        let _ = writeln!(self.file, "{}  {}", local_time_string(), msg);
        let _ = self.file.flush();
    }
}

/// 日志目录：正式版用 %LOCALAPPDATA%\pi-launcher\logs
pub fn log_dir() -> PathBuf {
    let base = std::env::var("LOCALAPPDATA")
        .map(PathBuf::from)
        .unwrap_or_else(|_| PathBuf::from("."));
    base.join("pi-launcher").join("logs")
}

/// 清理旧日志，仅保留最近 keep 份（文件名含时间戳，按名排序即可）
fn cleanup_logs(dir: &Path, keep: usize) -> io::Result<usize> {
    let mut logs: Vec<PathBuf> = fs::read_dir(dir)?
        .filter_map(|e| e.ok())
        .map(|e| e.path())
        .filter(|p| {
            p.file_name()
                .and_then(|n| n.to_str())
                .map(|n| n.starts_with(LOG_PREFIX) && n.ends_with(".log"))
                .unwrap_or(false)
        })
        .collect();
    logs.sort();
    let total = logs.len();
    let mut removed = 0usize;
    if total > keep {
        for p in &logs[..total - keep] {
            let _ = fs::remove_file(p);
            removed += 1;
        }
    }
    Ok(removed)
}

fn stamp_hex() -> String {
    let secs = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .unwrap_or_default()
        .as_secs();
    format!("{:012x}", secs)
}

/// 初始化全局日志器（两种模式都在 main 里最先调用）。
/// 返回日志文件完整路径。
pub fn init_logger() -> io::Result<String> {
    let logger = Logger::new()?;
    let path = logger.path.to_string_lossy().to_string();
    let _ = LOGGER.set(Mutex::new(logger));
    Ok(path)
}

/// 任意线程可用的日志输出（时间戳自动附加）
pub fn log(msg: &str) {
    if let Some(m) = LOGGER.get() {
        if let Ok(mut l) = m.lock() {
            l.write(msg);
        }
    }
}

/// 当前日志文件路径（未初始化时返回占位）
pub fn log_path() -> String {
    match LOGGER.get().and_then(|m| m.lock().ok()) {
        Some(l) => l.path.to_string_lossy().to_string(),
        None => "(日志未初始化)".to_string(),
    }
}

/// 行前时间戳：Windows 用系统本地时间；其它平台退化为 UTC
#[cfg(windows)]

    use windows_sys::Win32::System::SystemInformation::GetLocalTime;
    unsafe {
        let mut st = SYSTEMTIME {
            wYear: 0,

    }
}

#[cfg(not(windows))]
pub fn local_time_string() -> String {
    let secs = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .unwrap_or_default()
        .as_secs() as i64;
    let (y, m, d) = civil_from_days(secs.div_euclid(86_400));
    let s = secs.rem_euclid(86_400);
    format!(
        "{:04}-{:02}-{:02} {:02}:{:02}:{:02}",
        y,
        m,
        d,
        s / 3600,
        (s % 3600) / 60,
        s % 60
    )
}

/// Howard Hinnant 的 civil_from_days：unix 天数 → (年, 月, 日)
#[cfg(not(windows))]
fn civil_from_days(z: i64) -> (i64, u32, u32) {
    let z = z + 719_468;
    let era = z.div_euclid(146_097);
    let doe = z.rem_euclid(146_097);
    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
    let y = yoe + era * 400;
    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
    let mp = (5 * doy + 2) / 153;
    let d = (doy - (153 * mp + 2) / 5 + 1) as u32;
    let m = if mp < 10 { mp + 3 } else { mp - 9 } as u32;
    (if m <= 2 { y + 1 } else { y }, m, d)
}

/// --spin 指定 > 否则按「本机已有日志数」循环轮换
/// （每次启动产生一个新日志文件，日志数即启动次数，天然形成循环）
pub fn pick_anim(forced: Option<&str>) -> (usize, &'static str, &'static str) {
    let list = anim_pool();

    if let Some(name) = forced {
        if let Some(i) = list.iter().position(|(n, _)| *n == name) {
            return (i, list[i].0, list[i].1);
        }
        println!(
            "  ⚠ 未知动画名 '{}'，改用循环轮换（可选：{}）",
            name,
            list.iter().map(|(n, _)| *n).collect::<Vec<_>>().join(", ")
        );
    }

    let runs = existing_run_count();
    let i = runs % list.len();
    (i, list[i].0, list[i].1)
}

fn existing_run_count() -> usize {
    fs::read_dir(log_dir())
        .map(|it| {
            it.filter_map(|e| e.ok())
                .filter(|e| {
                    e.file_name()
                        .to_str()
                        .map(|n| n.starts_with(LOG_PREFIX) && n.ends_with(".log"))
                        .unwrap_or(false)
                })
                .count()
        })
        .unwrap_or(0)
}

struct Notify {
    mp: Option<MultiProgress>,
    plain_text: bool,
    quiet: bool,
}

static NOTIFY: OnceLock<Notify> = OnceLock::new();

/// 后台线程（supervisor 等）用的提示：打印（安静模式除外）+ 日志
pub fn notify(msg: &str) {
    log(msg);
    match NOTIFY.get() {
        None => println!("{}", msg),
        Some(n) if n.quiet => {}
        Some(n) if n.plain_text => println!("{}", msg),
        Some(n) => {
            if let Some(mp) = &n.mp {
                mp.println(msg).ok();
            } else {
                println!("{}", msg);
            }
        }
    }
}

                overall.set_style(
                    ProgressStyle::with_template("  整体进度 {bar:26.cyan/green} {percent:>3}%")
                        .unwrap()
                        .progress_chars("━╸─"),
                );

                // supervisor 等后台线程需要同一条 MultiProgress 来安全输出
                let _ = NOTIFY.set(Notify {
                    mp: Some(mp.clone()),
                    plain_text: false,
                    quiet: false,
                });

                Ui {
                    kind: Kind::Interactive {
                        mp,
                        spin_chars: chars,
                    },
                    steps_total,
                    anim_name: Some(name),
                    anim_index: Some(idx),
                    overall: Some(overall),
                    overall_base: 0,
                    current_bar: None,
                    current_det: false,
                    current_idx: 0,
                    step_started: None,
                }
            }
        }
    }

    /// 本次动画名（交互模式才有；供 banner 展示轮换结果）
    pub fn anim_label(&self) -> Option<(usize, &'static str)> {
        match (&self.anim_index, &self.anim_name) {
            (Some(i), Some(n)) => Some((*i, *n)),
            _ => None,
        }
    }

    /// 通用一行（含 ✓/✗/⚠/ℹ 前缀均由调用方给出）：打印（安静模式除外）+ 落盘
    pub fn message(&self, msg: &str) {
        log(msg);
        match &self.kind {
            Kind::Interactive { mp, .. } => {
                mp.println(msg).ok();
            }
            Kind::Text => println!("{}", msg),
            Kind::Quiet => {}
        }
    }

    /// stderr 摘要打印（供失败步骤复用）：缩进显示每行
    pub fn print_stderr_summary(&self, stderr: &str) {
        if stderr.is_empty() {
            return;
        }
        self.message("  错误摘要：");
        for line in stderr.lines() {
            self.message(&format!("    {}", line));
        }
    }

    /// 步骤开始：打印步骤标题、建 bar（det → 确定进度条；spin → 右侧转圈）并记日志
    pub fn step_start(&mut self, idx: usize, label: &str, det: bool) {
        let total = self.steps_total;
        log(&format!("[STEP {}/{}] 开始：{}", idx, total, label));

        self.current_idx = idx;
        self.current_det = det;
        self.step_started = Some(Instant::now());
        self.current_bar = None;

        match &self.kind {
            Kind::Interactive { mp, .. } => {
                mp.println(format!("\n  [{}/{}] {}", idx, total, label))
                    .ok();

                let bar = mp.add(ProgressBar::new(100));
                let style = if det {
                    det_style()
                } else {
                    spin_style(self.spin_chars())
                };
                bar.set_style(style);
                bar.set_message(if det { "等待中…" } else { "执行中…" });
                bar.enable_steady_tick(std::time::Duration::from_millis(90));
                self.current_bar = Some(bar);
            }
            Kind::Text => {
                println!("\n  [{}/{}] {}", idx, total, label);
            }
            Kind::Quiet => {}
        }
    }

    /// 更新当前步骤的 bar 右侧消息
    pub fn step_message(&mut self, msg: &str) {
        if let Kind::Interactive { .. } = &self.kind {
            if let Some(bar) = &self.current_bar {
                bar.set_message(msg.to_string());
            }
        }
    }

    /// 确定进度（0.0-1.0）：更新当前步骤 bar + 整体进度条按窗口爬升
    pub fn step_progress(&mut self, frac: f64) {
        if !self.current_det {
            return;
        }
        let per = 100u64 / self.steps_total.max(1) as u64;
        if let Some(bar) = &self.current_bar {
            bar.set_position((frac * 100.0) as u64);
        }
        if let Some(overall) = &self.overall {
            let creep = (frac.clamp(0.0, 1.0) * per as f64) as u64;
            overall.set_position((self.overall_base + creep).min(self.overall_base + per));
        }
    }

    /// 步骤成功：清 bar、打印 ✓、日志带耗时；整体进度推进一个步距
    pub fn step_ok(&mut self, msg: &str) {
        let secs = self.step_secs();
        let idx = self.current_idx;
        let per = 100u64 / self.steps_total.max(1) as u64;

        log(&format!(
            "[STEP {}/{}] 完成：{}（{:.1}s）",
            idx, self.steps_total, msg, secs
        ));

        if let Some(bar) = self.current_bar.take() {
            bar.finish_and_clear();
        }
        self.step_started = None;

        match &self.kind {
            Kind::Interactive { mp, .. } => {
                mp.println(format!("  ✓ {}", msg)).ok();
                if let Some(overall) = &self.overall {
                    self.overall_base = ((self.overall_base / per) + 1) * per;
                    overall.set_position(self.overall_base);
                }
            }
            Kind::Text => println!("  ✓ {}（{:.1}s）", msg, secs),
            Kind::Quiet => {}
        }
    }

    /// 步骤失败：清 bar、打印 ✗、落盘
    pub fn step_fail(&mut self, msg: &str) {
        let secs = self.step_secs();
        let idx = self.current_idx;
        log(&format!(
            "[STEP {}/{}] 失败：{}（{:.1}s）",
            idx, self.steps_total, msg, secs
        ));

        if let Some(bar) = self.current_bar.take() {
            bar.finish_and_clear();
        }
        self.step_started = None;

        match &self.kind {
            Kind::Interactive { mp, .. } => {
                mp.println(format!("  ✗ {}", msg)).ok();
            }
            Kind::Text => println!("  ✗ {}", msg),
            Kind::Quiet => {}
        }
    }

    /// 启动 pi CLI 前清掉所有剩余渲染（TUI 与 bar 重绘会互相打架）
    pub fn overall_finish(&mut self) {
        if let Some(overall) = self.overall.take() {
            overall.finish_and_clear();
        }
    }

    fn spin_chars(&self) -> &'static str {
        match &self.kind {
            Kind::Interactive { spin_chars, .. } => spin_chars,
            _ => "",
        }
    }

    fn step_secs(&self) -> f64 {
        self.step_started
            .map(|s| s.elapsed().as_secs_f64())
            .unwrap_or(0.0)
    }
}

fn spin_style(spin_chars: &'static str) -> ProgressStyle {
    // 右侧固定位置：{wide_msg} 把残余宽度撑满，将转圈 + 已耗时推到行末
    ProgressStyle::with_template("{wide_msg} {spinner:.green} {elapsed:>4}")
        .unwrap()
        .tick_chars(spin_chars)
}

fn det_style() -> ProgressStyle {
    ProgressStyle::with_template("{wide_msg} {bar:14.cyan/blue} {percent:>3}% eta:{eta:>3}")
        .unwrap()
        .progress_chars("━╸─")
}

fn parse_args() -> Args {
    let mut args = Args {
        mode: Mode::Normal,
        spin: None,
        force_tty: false,
    };

    let mut it = std::env::args().skip(1);
    while let Some(a) = it.next() {
        match a.as_str() {
            "--quiet" | "-q" | "--auto" => args.mode = Mode::Quiet,
            "--force-tty" => args.force_tty = true, // 验证用法：强制走动画渲染路径
            "--spin" => {
                if let Some(name) = it.next() {
                    args.spin = Some(name);
                }
            }
            "-h" | "--help" => {
                print_help();
                process::exit(0);
            }
            other => {
                println!("  未知参数 '{}'（见 --help）", other);
                process::exit(1);
            }
        }
    }
    args
}

fn boot_pipeline(ui: &mut Ui, host: &str, port: u16) -> Result<WebStart, i32> {
    if step_preflight(ui).is_err() {
        return Err(1);
    }
    if step_pi_update(ui).is_err() {
        return Err(2);
    }
    if step_pi_web_upgrade(ui, host, port).is_err() {
        return Err(3);
    }
    match step_pi_web(ui, host, port) {
        Ok(ws) => Ok(ws),
        Err(_e) => Err(4),
    }
}

fn run_quiet(host: String, port: u16) -> i32 {
    let mut ui = Ui::init(Mode::Quiet, false, None, steps::TOTAL_STEPS);

    println!("  pi-launcher（--quiet）：静默启动中，完成后窗口自动关闭。");
    println!("    日志：{}", ui::log_path());

    match boot_pipeline(&mut ui, &host, port) {
        Ok(ws) => {
            if let WebStart::Spawned(child) = ws {

            ui::log("pi CLI 已按安静模式策略跳过。");
            ui::log("全部步骤完成；退出码 0（pi-web 若已启动，独立后台运行）");
            0
        }
        Err(code) => {
            ui::log(&format!("最终退出码：{}", code));
            code
        }
    }
}

fn run_normal(args: &Args, host: String, port: u16) -> i32 {
    let mut ui = Ui::init(
        Mode::Normal,
        args.force_tty,
        args.spin.as_deref(),
        steps::TOTAL_STEPS,
    );
    ui::log("启动模式：正常（pi CLI 交互 + 结束停留）");
    print_banner(&ui);

    match boot_pipeline(&mut ui, &host, port) {
        Err(code) => {
            ui.overall_finish();
            wait_and_exit(code);
        }
        Ok(WebStart::Spawned(child)) => {
            spawn_supervisor(child, host.clone(), port);
        }
        Ok(WebStart::AlreadyRunning) => {}
    }

    if step_pi_cli(&mut ui).is_err() {
        wait_and_exit(5);
    }

    ui.message(&format!("  📝 本次启动日志：{}", ui::log_path()));
    ui.message("  ✓ 全部步骤完成。");
    wait_and_exit(0)
}

    }
    ui.message("");
    ui.message("  将依次执行：");
    ui.message("    1. 环境预检");
    ui.message("    2. pi update --all");
    ui.message("    3. 检查并升级 pi-web");

// ────────────────────────────────────────────────────────────────
// 结束停留：按 Enter 立即关闭；30 秒无输入自动关闭（含失败停留路径）
// ────────────────────────────────────────────────────────────────

fn wait_and_exit(code: i32) -> ! {
    pause_before_exit(code);
    process::exit(code);
}

fn pause_before_exit(exit_code: i32) {
    let term = console::Term::stdout();
    if !term.is_term() {
        ui::log(&format!("非交互终端，直接关闭；退出码 {}", exit_code));
        println!("\n  （非交互终端，直接关闭）");
        return;
    }

    println!();
    let pressed = Arc::new(AtomicBool::new(false));
    {
        let pressed = Arc::clone(&pressed);
        let secs = WAIT_AUTOCLOSE_SECS;
        thread::spawn(move || {
            for remain in (1..=secs).rev() {
                if pressed.load(Ordering::SeqCst) {
                    return;
                }
                print!("\r  按 Enter 键关闭此窗口 · {} 秒后自动关闭 ...   ", remain);
                let _ = std::io::Write::flush(&mut std::io::stdout());
                thread::sleep(Duration::from_secs(1));
            }
            if !pressed.load(Ordering::SeqCst) {
                ui::log("30 秒无输入，自动关闭");
                process::exit(0);
            }
        });
    }

    let mut line = String::new();
    let _ = std::io::stdin().read_line(&mut line);
    pressed.store(true, Ordering::SeqCst);
    println!();
}

// ────────────────────────────────────────────────────────────────
// pi-web 守护线程：崩溃自动重启（最多 MAX_RESTARTS 次）+ HTTP 健康告警
// ────────────────────────────────────────────────────────────────

fn spawn_supervisor(mut child: Child, host: String, port: u16) {
    thread::spawn(move || {
        let mut restarts = 0u32;
        let mut unhealthy = 0u32;
        let mut ticks = 0u64;

        loop {
            thread::sleep(Duration::from_millis(2000));
            ticks += 1;

            match child.try_wait() {
                Ok(Some(status)) => {
                    ui::notify(&format!(
                        "  ⚠ [supervisor] 侦测到 pi-web 已退出（{}）",
                        status
                    ));
                    if restarts >= MAX_RESTARTS {
                        ui::notify(&format!(
                            "  ✗ [supervisor] 已重启 {} 次仍未稳定，停止守护；请手动执行 `pi-web` 排查。",
                            restarts
                        ));
                        return;
                    }
                    restarts += 1;
                    ui::notify(&format!(
                        "  ↻ [supervisor] 正在重启 pi-web（第 {} 次）…",
                        restarts
                    ));

                    match spawn_detached("pi-web", &[]) {
                        Ok(mut c) => {
                            match command_runner::wait_ready(
                                &mut c,
                                &host,
                                port,
                                30,
                                Some(|_: f64| {}),
                            ) {
                                command_runner::StartupOutcome::Up => {
                                    ui::notify("  ✓ [supervisor] pi-web 已重启并就绪");
                                    child = c;
                                    unhealthy = 0;
                                }
                                other => {
                                    ui::notify(&format!(
                                        "  ⚠ [supervisor] pi-web 重启后未就绪：{:?}",
                                        other
                                    ));
                                    child = c;
                                }
                            }
                        }
                        Err(e) => {
                            ui::notify(&format!("  ✗ [supervisor] 无法重启 pi-web：{}", e));
                            return;
                        }
                    }
                }
                Ok(None) => {
                    // 每 ~10 秒做一次 HTTP 健康探测
                    if ticks.is_multiple_of(5) {
                        if command_runner::probe_http(&host, port) {
                            unhealthy = 0;
                        } else {
                            unhealthy += 1;
                            if unhealthy >= 3 {
                                ui::notify(&format!(
                                    "  ⚠ [supervisor] pi-web 进程存活但 HTTP 无响应（连续 {} 次）",
                                    unhealthy
                                ));
                            }
                        }
                    }
                }
                Err(_) => {}
            }
        }
    });
}

11设计说明

为什么所有输出都走 ui.rs

同一段步骤逻辑要在三种环境正确工作：全功能动画终端、CI 管道、计划任务。分散的 println! 无法统一三种表现，也无法保证「不打印输出但日志照写」。统一输出层让步骤代码完全不感知模式差异。

为什么 CLI 前要清掉进度条

Pi CLI 是全屏交互界面：任何还在重绘的 bar 都会破坏它的画面。进入步骤 5 前先整体 finish_and_clear，之后 supervisor 消息退化为普通 println（此时打印是安全的）。

为什么不伪造进度

npm 系命令的真实进度不可观测，中间过程一律只显示「已耗时」与转圈动画，不虚构百分比。唯一确定时长的场景是「等待 pi-web HTTP 就绪」（60 秒时间轴），其比例条与真实耗时一致。

升级策略（已定稿）

文件锁处理 pi-web 正在运行 → 本次不升级，按现版继续；升级撞锁（EBUSY / -4082 / "busy or locked"）→ 警告后按现版继续。两者都保留可观测性（警告 + 日志）。想完成升级：pi-web stop → 再跑一次 launcher。

supervisor 为何两种模式都在

安静模式下没有人盯着窗口，pi-web 却一样会出问题——因此守护与重启循环不随渲染模式变化，只是消息都收敛到日志。

为什么退出码细分

计划任务的自动化不能依赖人眼：退出码细分到每个可失败步骤，日志记录到该退出码的完整路径，单独看一次日志就知道挂在哪一步。
